        \scriptsize

        \[
          \mathbf{\min_{\mathbf{S}} \; T_{\text{race}}(\mathbf{S}) = \sum_{k=1}^{K+1} \sum_{a=1}^{l_k} t_{\text{lap}}(c_k, a, i) + K \cdot T_{\text{pit}}}

      \end{alertblock}

        \scriptsize
        \begin{enumerate}
          \item \textbf{Bảo toàn tổng số vòng:} $\sum_{k=1}^{K+1} l_k = N_{\text{total}} = 53\,\text{vòng (Monza)}$.
          \item \textbf{Quy tắc hợp chất FIA:} $\big|\{c_1, \dots, c_{K+1}\}\big| \ge 2 \implies \Phi_{\text{FIA}} = 10^5\,\text{s}$.
          \item \textbf{Giới hạn an toàn tuổi thọ lốp:} $l_k \le L_{\max}(c_k)$ (\badgeSoft $\le 22$, \badgeMed $\le 35$, \badgeHard $\le 48$).
        \end{enumerate}
      \end{block}
    \end{column}
  \end{columns}

    \begin{column}{0.50\textwidth}
      \begin{alertblock}{\textbf{Đóng Góp 1: Đạo Hàm Tốc Độ Suy Thoái Biên $\nabla t_{\text{deg}}$}}
        \scriptsize
        \[
          \mathbf{\nabla t_{\text{deg}}(c, a) = \frac{\partial \Delta t_{\text{deg}}}{\partial a} = d_1(c) + 2 d_2(c) a}
        \]
        \begin{itemize}
          \item $\nabla^2 t = 2d_2 > 0 \implies$ Hàm lồi nghiêm ngặt, tồn tại duy nhất một nghiệm cực tiểu.
          \item Cung cấp \textbf{véc-tơ gradient hướng dẫn} cơ chế sửa nghiệm vi phạm ràng buộc (Physics-guided Repair).
        \end{itemize}
      \end{alertblock}

      \begin{exampleblock}{\textbf{Đóng Góp 2: Dạng Đóng Giải Tích Faulhaber $O(1)$}}
        \scriptsize
        \[
          \mathbf{D_{\text{stint}}(c, L) = d_1 \frac{L(L+1)}{2} + d_2 \frac{L(L+1)(2L+1)}{6}}
        \]
        \begin{itemize}
          \item Chuyển đổi vòng lặp tuần tự $O(L)$ thành phép tính giải tích \textbf{$O(1)$}.
          \item \textbf{Tăng tốc tính toán gấp 50 lần}, cho phép thuật toán Quy hoạch động duyệt 30.088 trạng thái trong $< 0.1\,\text{s}$.
        \end{itemize}
      \end{exampleblock}
