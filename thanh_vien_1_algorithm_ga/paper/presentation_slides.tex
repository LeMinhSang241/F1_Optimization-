        $$\mathbf{\min_{\mathbf{S}} \; T_{\text{race}}(\mathbf{S}) = \sum_{k=1}^{K+1} \sum_{a=1}^{l_k} t_{\text{lap}}(c_k, a, i) + K \cdot T_{\text{pit}}}$$

          \item \textbf{Ngưỡng an toàn vách lốp:} $l_k \le L_{\max}(c_k)$ (\badgeSoft $\le 22$, \badgeMed $\le 35$, \badgeHard $\le 48$).
        \end{enumerate}

    \begin{column}{0.50\textwidth}
      \begin{alertblock}{\textbf{Khám Phá 1: Đạo Hàm Tốc Độ Mòn Biên $\nabla t_{\text{deg}}$}}
        \scriptsize
        $$\mathbf{\nabla t_{\text{deg}}(c, a) = \frac{\partial \Delta t_{\text{deg}}}{\partial a} = d_1(c) + 2 d_2(c) a}$$
        \begin{itemize}
          \item $\nabla^2 t = 2d_2 > 0 \implies$ Tính lồi nghiêm ngặt, 1 cực tiểu duy nhất.
          \item Đóng vai trò \textbf{"la bàn" gradient vật lý} để sửa nghiệm di truyền.
        \end{itemize}
      \end{alertblock}

      \begin{exampleblock}{\textbf{Khám Phá 2: Đột Phá Tổng Đóng Faulhaber $O(1)$}}
        \scriptsize
        $$\mathbf{D_{\text{stint}}(c, L) = d_1 \frac{L(L+1)}{2} + d_2 \frac{L(L+1)(2L+1)}{6}}$$
        \begin{itemize}
          \item Biến toàn bộ vòng lặp $O(L)$ thành phép tính giải tích \textbf{$O(1)$}.
          \item \textbf{Tăng tốc tính toán gấp 50 lần}, giúp DP giải 30.088 trạng thái trong $< 0.1\,\text{s}$.
        \end{itemize}
      \end{exampleblock}
    \end{column}
